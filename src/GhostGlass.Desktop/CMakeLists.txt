cmake_minimum_required(VERSION 3.25)
project(GhostGlass VERSION 0.1.0 LANGUAGES CXX)
add_executable(GhostGlass WIN32 main.cpp panel.cpp)
target_compile_features(GhostGlass PRIVATE cxx_std_20)
target_compile_definitions(GhostGlass PRIVATE UNICODE _UNICODE)
target_link_libraries(GhostGlass PRIVATE d3d11 d2d1 dxgi dcomp user32 shell32 gdi32 windowsapp comdlg32)
if(MSVC)
 target_compile_options(GhostGlass PRIVATE /W4 /EHsc /permissive- /utf-8)
endif()
